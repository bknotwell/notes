Generating and publishing notes is done with a combination of emacs
lisp, shell commands and [[https://www.gnu.org/software/make/][gmake]]:

#+begin_src makefile
  EMACS=/Applications/Emacs.app/Contents/MacOS/Emacs
  SCRIPT=~/roam/index-content.el

  define PUBLISH
      ssh $(1) rm /var/www/roam/*.html /var/www/roam/*.svg
      scp *.html $(1):/var/www/roam
      scp *.svg $(1):/var/www/roam
      scp t.db $(1):/var/www/roam
      ssh $(1) sudo  cp /var/www/roam/t.db /srv/shiny-server/notes-search/t.db 
      ssh $(1) sudo chown $(2) /var/www/roam/*.html /var/www/roam/*.svg
  endef

  all: publish-content

  publish-content: publish-myfree1 publish-beastie

  publish-beastie: gen-content
      $(call PUBLISH, beastie, www)

  publish-myfree1: gen-content
      $(call PUBLISH, myfree1, nginx)

  gen-content:
      rm t.db
      $(EMACS) -q  --script $(SCRIPT)
      mv ../roam/*.svg .

  .PHONY: gen-content

#+end_src
