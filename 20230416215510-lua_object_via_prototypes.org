Lua's use of prototype based inheritance relies on the [[http://www.lua.org/pil/13.4.1.html][__index]]
metamethod:

#+begin_src lua
animal = { __index = {
      jump = function(x) return(x.height .. ' meter') end
  }
}

flea = {
   init=function()
       x = {}
       setmetatable(x, animal)
       x.height = .005
       return(x)
   end
}

human = {
   init=function()
       x = {}
       setmetatable(x, animal)
       x.height = 1
       return(x)
   end
}

f1 = flea:init()
h1 = human:init()
print(f1:jump())
print(h1:jump())
#+end_src
