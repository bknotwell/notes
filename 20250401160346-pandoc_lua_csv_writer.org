The code below creates a custom writer that allows data from the
[[id:71706ecc-980e-468d-8eca-121e08dee6ae][Pandoc lua table filter]] to be written as a CSV.

#+begin_src lua
res = 'Filename,Rownum\n'
rownum = 1
cb = {
    Str = function(el)
        res = res .. pandoc.utils.stringify(el) .. ',' .. rownum .. '\n'
        rownum = rownum + 1
    end
}

function Writer(doc, opts)
    doc.blocks:walk(cb)
    return res
end
#+end_src
