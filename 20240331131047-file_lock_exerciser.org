The code below was used to exercise file locks on a V4 NFS filesystem.
It compiles w/o warnings on FreeBSD and has fun on the AMD64 and ARM64
architectures.

#+begin_src C
  #include <stdio.h>
  #include <unistd.h>
  #include <stdlib.h>
  #include <fcntl.h>
  #include <sys/file.h>
  #include <errno.h>

  int main(void) {

      int s = open("/efs/mylock.lock", O_WRONLY | O_CREAT, 0666);

      if(s == -1)
          return(fprintf(stderr, "Failed opening lockfile: %d\n", errno),
                 EXIT_FAILURE);
      for( ; ; ) {
          if(flock(s, LOCK_EX | LOCK_NB) == -1) {
              printf("Lock acquisition failed %s\n",
                     errno == EWOULDBLOCK ? "normal" : "true failure");
              usleep(100000);

          } else {
              printf("Lock acquisition succeeded\n");
              usleep(200000);
              if(flock(S, LOCK_UN) == -1)
                  return(fprintf(stderr, "Failed unlocking file: %d\n", errno),
                         EXIT_FAILURE);
          }
      }
  }

#+end_src
