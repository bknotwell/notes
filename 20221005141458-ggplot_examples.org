* Examples
** Scatterplot with dates and scales

#+begin_src R
  library(ggplot2)
  library(scales)

  data <- read.csv('t.csv')
  data$X_time <- as.POSIXct(data$X_time, format='%Y-%m-%dT%H:%M:%S.000+0000')

  options(scipen=999)  # disable scientific notation

  ggplot(data, aes(x=X_time, y=count * 100000)) + geom_point() +
      ggtitle('System messages/hour'),\ + theme(plot.title = element_text(hjust = 0.5)) +
      scale_y_continuous(labels=comma) + ylab('Count') + xlab('Date')                                        
#+end_src
