Getting certificate information from a local certificate:
=openssl x509 -in filename.crt -text=

From a remote server:

#+begin_src sh
  openssl s_client -connect ip:port 2>/dev/null | \
    sed '/-----BEGIN CERTIFICATE-----/,/-----END CERTIFICATE-----/!d' | \
    openssl x509 -text
#+end_src
