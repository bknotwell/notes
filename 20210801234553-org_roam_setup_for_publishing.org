#+title: Org-roam setup for publishing
#+OPTIONS: ^:nil author:nil html-postamble:nil
#+SETUPFILE: ./latexcss.theme
#+HTML_HEAD_EXTRA: <style> pre.src { background-color: black; color: white; }</style>


The configuration snippet below will setup org-roam and get it ready to publish through =org-publish=:
#+begin_src emacs-lisp
  (use-package org-roam
    :ensure t
    :init (setq org-roam-v2-ack 't)
    :custom (org-roam-directory "~/roam")
    :bind (("C-c n l" . org-roam-buffer-toggle)
	   ("C-c n f" . org-roam-node-find)
	   ("C-c n i" . org-roam-node-insert))
    :config
    (org-roam-setup))

  (defun roam-sitemap (title list)
    (concat "#+OPTIONS: ^:nil author:nil html-postamble:nil\n"
	    "#+SETUPFILE: ./latexcss.theme\n"
	    "#+TITLE: " title "\n\n"
	    (org-list-to-org list) "\nfile:sitemap.svg"))

  (add-hook 'org-roam-graph-generation-hook
	    (lambda (dot svg) (copy-file svg "~/roam-export/sitemap.svg" 't)))

  (defun roam-publication-wrapper (plist filename pubdir)
    (org-roam-graph)
    (org-html-publish-to-html plist filename pubdir))

  (setq org-publish-project-alist
	'(("roam"
	   :base-directory "~/roam"
	   :auto-sitemap t
	   :sitemap-function roam-sitemap
	   :sitemap-title "Roam notes"
	   :publishing-function roam-publication-wrapper
	   :publishing-directory "~/roam-export"
	   :section-number nil
	   :table-of-contents nil
	   :style "<link rel=\"stylesheet\" href=\"../other/mystyle.cs\" type=\"text/css\">")))

  (setq org-roam-capture-templates
	'(("d" "default" plain "%?"
	   :if-new (file+head "%<%Y%m%d%H%M%S>-${slug}.org"
			      "#+title: ${title}\n#+OPTIONS: ^:nil author:nil html-postamble:nil\n#+SETUPFILE: ./latexcss.theme\n\n\n[[file:sitemap.org][Sitemap]]")
	   :unnarrowed t)))

  (defun org-roam-custom-link-builder (node)
    (let ((file (org-roam-node-file node)))
      (concat (file-name-base file) ".html")))
#+end_src
