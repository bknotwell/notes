The function =zzz= iterates over a list of four-element items and opens
up the file, searches for a match, inserts the generated text and
saves the file.  The files are left open for further editing.

#+begin_src emacs-lisp
  (defun zzz (elt)
    (let ((file (car elt))
          (repo (cadr elt))
          (stanza (format (concat "   cst: %d\n"
                                  "   cis: %s\n")
                          (caddr elt) (cadddr elt))))
      (progn
        (find-file file)
        (goto-char (point-min))
        (if (not (equal (search-forward (concat repo "\n") nil t) nil))
            (progn (insert stanza) (save-buffer))))))

  (mapcar #'zzz incoming-list)
#+end_src
