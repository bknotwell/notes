#+begin_src R :session R-russian-casualties :var filename="./russia_losses_personnel.csv" :exports both :results output graphics file :file russian-losses.svg
  data <- read.csv(filename)
  data$Date <- as.Date(data$date)
  data$numdate <- as.numeric(data$Date, origin='1970-01-01')
  plot(data$Date, data$personnel, main='Russian personnel losses',
       xlab='Date', ylab='Count', pch=20)
  m <- lm(personnel ~ Date, data = data)
  abline(m)
  interval <- floor((data$Date[nrow(data)] - data$Date[1]) / 4)
  l <- c(data$Date[1], data$Date[1] + interval,
         data$Date[1] + 2 * interval, data$Date[nrow(data)])
  axis(1, at=seq(1, nrow(data), as.numeric((-interval)+1)), labels=l)
#+end_src

** Both predictions with confidence intervals

Show confidence intervals instead of specific numbers.

#+begin_src R :session R-russian-casualties :exports both
  m4 <- lm(personnel ~ Date, data)
  res <- round(predict(m4, newdata=data.frame(Date=as.Date('2023-02-22',
                                                       origin='1970-01-01')),
                       interval='confidence', level=0.95))

   matrix(c(c('expected', 'lwr', 'upr'),
          as.integer(round(res / 100) * 100)), byrow=TRUE, nrow=2)
#+end_src
