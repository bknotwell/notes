Disabling echo when capturing a password in a shell script:

#+begin_src sh
# setup a safe settings reset
TERMINALSETTINGS=$(stty -g) || exit 1
trap 'stty "${TERMINALSETTING}"' EXIT INT TERM

# disable echoing the keyboard input and read the password
stty -echo || exit 2
printf "Password: "
read -r password
#+end_src
